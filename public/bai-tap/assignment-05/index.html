Bài tập 01

Quản lý timer

Nếu bỏ if (timerId) { clearTimeout(timerId); }, mỗi lần bấm tạo một timer riêng và không hủy timer đang chờ. Mỗi callback thực thi khoảng 3 giây sau lần bấm tương ứng, nên bấm nhiều lần nhanh sẽ có nhiều lần thực thi. Biến timerId chỉ bị gán ID mới, không tự hủy timer cũ. Giữ clearTimeout thì các lần bấm trong lúc chờ sẽ đặt lại thời gian, và chỉ callback của lần bấm cuối còn chạy sau 3 giây. Nếu chỉ bấm một lần, hai phiên bản cho kết quả như nhau. Trong ví dụ gốc, mọi callback gán cùng nội dung nên nhìn vào dòng trạng thái có thể khó nhận ra nhiều lần thực thi; demo thêm bộ đếm để minh họa.

Chạy bài 1 →Tải mã nguồn
Xem mã hoàn chỉnh
// Hai phiên bản chỉ khác việc hủy timer đang chờ trước khi đặt timer mới.
const cacViDu = [
  { suffix: "", huyTimerCu: true },
  { suffix: "-no-clear", huyTimerCu: false }
];
const datLai = [];
for (const viDu of cacViDu) {
  let timerId = null;
  let soLan = 0;
  const dangCho = new Set();
  const status = document.getElementById(`status${viDu.suffix}`);
  const count = document.getElementById(`count${viDu.suffix}`);
  const log = document.getElementById(`log${viDu.suffix}`);
  document.getElementById(`start${viDu.suffix}`).onclick = () => {
    if (viDu.huyTimerCu && timerId) {
      clearTimeout(timerId);
      dangCho.delete(timerId);
    }
    status.textContent = "Status: Timer started (3s)";
    const id = setTimeout(() => {
      dangCho.delete(id);
      soLan += 1;
      status.textContent = "✅ Action executed!";
      count.textContent = soLan;
      const dong = document.createElement("li");
      dong.textContent = `Callback thực thi lần ${soLan}.`;
      log.append(dong);
    }, 3000);
    timerId = id;
    dangCho.add(id);
  };
  datLai.push(() => {
    for (const id of dangCho) clearTimeout(id);
    dangCho.clear();
    timerId = null;
    soLan = 0;
    count.textContent = "0";
    status.textContent = "Status: Waiting...";
    log.replaceChildren();
  });
}
document.getElementById("reset").onclick = () => datLai.forEach(ham => ham());

Bài tập 02

Bỏ từ khóa await

const msg = wait(1000) gán đối tượng Promise đang chờ vào msg thay vì giá trị chuỗi đã resolve. Lệnh out.textContent = msg chạy ngay và chuyển Promise thành chuỗi [object Promise], nên dòng Waiting hầu như không kịp hiển thị. Một giây sau Promise resolve thành ✅ Done nhưng không có câu lệnh nào chờ rồi cập nhật giao diện, nên trang vẫn hiện [object Promise]. Có await, hàm bất đồng bộ tạm dừng đến khi Promise hoàn thành rồi nhận chuỗi ✅ Done và gán vào textContent; điều này không chặn luồng giao diện.

Chạy bài 2 →Tải mã nguồn
Xem mã hoàn chỉnh
function wait(ms) {
  return new Promise(resolve => setTimeout(() => resolve("✅ Done"), ms));
}

async function runDemo() {
  const out = document.getElementById("out");
  out.textContent = "⏳ Waiting...";
  const msg = await wait(1000);
  out.textContent = msg;
}

async function runWithoutAwait() {
  const out = document.getElementById("out-no-await");
  out.textContent = "⏳ Waiting...";
  const msg = wait(1000);
  out.textContent = msg;
}

document.getElementById("btn").onclick = runDemo;
document.getElementById("btn-no-await").onclick = runWithoutAwait;

Bài tập 03

Promise tải người dùng

Với ID 1: hiện Đang tải... rồi User: An sau khoảng một giây. Với ID -1: hiện Đang tải... rồi Lỗi: ID không hợp lệ. await biến Promise bị reject thành lỗi tại vị trí chờ để catch xử lý.

Đáp án các chỗ trống bài 3
ÔĐáp ánÝ nghĩa
1resolve, rejectHai callback xử lý thành công và lỗi của Promise.
2resolveHoàn thành Promise với object người dùng.
3rejectHoàn thành Promise với Error khi ID không hợp lệ.
4asyncCho phép dùng await bên trong showUser.
5tryBắt lỗi từ Promise bị reject bằng catch.
6awaitChờ fetchUser và nhận giá trị user.
Chạy bài 3 →Tải mã nguồn
Xem mã hoàn chỉnh
function fetchUser(id) {
  // (1) Hai callback của Promise.
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (id > 0) {
        // (2) Hoàn thành thành công.
        resolve({ id: id, name: "An" });
      } else {
        // (3) Hoàn thành với lỗi.
        reject(new Error("ID không hợp lệ"));
      }
    }, 1000);
  });
}

// (4) Hàm bất đồng bộ.
async function showUser(id) {
  const result = document.getElementById("result");
  result.textContent = "Đang tải...";
  // (5) Bắt lỗi Promise bị reject.
  try {
    // (6) Chờ giá trị thành công.
    const user = await fetchUser(id);
    result.textContent = `User: ${user.name}`;
  } catch (err) {
    result.textContent = `Lỗi: ${err.message}`;
  }
}

document.getElementById("ok").onclick = () => showUser(1);
document.getElementById("fail").onclick = () => showUser(-1);

Bài tập 04

Đếm ngược hủy đơn hàng

Khi đặt hàng, trang đếm từ 5 giây xuống. Hết giờ thì Promise resolve với Đơn hàng đã được xác nhận; bấm Hủy trước khi hết giờ thì clearInterval và reject với Error Bạn đã hủy đơn. answer.js là bản điền đúng khung đề. Demo chạy bổ sung khóa nút Đặt hàng trong lúc đang đếm, rồi mở lại sau khi xác nhận/hủy, để tránh nhiều interval dùng chung một intervalId.

Đáp án các chỗ trống bài 4
ÔĐáp ánÝ nghĩa
1setIntervalGọi callback mỗi 1000 ms.
2left--Giảm số giây còn lại đi một.
3clearIntervalDừng timer lặp khi hết giờ.
4resolveXác nhận đơn khi hết thời gian hủy.
5rejectBáo lỗi Bạn đã hủy đơn khi bấm Hủy.
Chạy bài 4 →Tải mã nguồn
Xem mã hoàn chỉnh
const status = document.getElementById("status");
let intervalId = null;

function countdown(seconds) {
  return new Promise((resolve, reject) => {
    let left = seconds;
    status.textContent = `Còn ${left}s để hủy...`;
    // (1) Lặp mỗi 1000 ms.
    intervalId = setInterval(() => {
      // (2) Giảm biến đếm.
      left--;
      status.textContent = `Còn ${left}s để hủy...`;
      if (left === 0) {
        // (3) Dừng lặp; (4) hoàn thành thành công.
        clearInterval(intervalId);
        resolve("Đơn hàng đã được xác nhận");
      }
    }, 1000);
    document.getElementById("cancel").onclick = () => {
      clearInterval(intervalId);
      // (5) Hoàn thành với lỗi khi hủy.
      reject(new Error("Bạn đã hủy đơn"));
    };
  });
}

document.getElementById("order").onclick = async () => {
  try {
    const msg = await countdown(5);
    status.textContent = `${msg}`;
  } catch (err) {
    status.textContent = `${err.message}`;
  }
};

Bài tập 05

Tra cứu thời tiết qua hai API

Gọi geocoding để lấy latitude/longitude rồi dùng hai tọa độ gọi forecast với current_weather=true. encodeURIComponent giữ tên thành phố an toàn trong query string. Bản chạy giữ bố cục mẫu, bổ sung kiểm tra response.ok, khóa nút khi tải và tạo DOM bằng textContent để nội dung API không bị diễn giải thành HTML. URL trong PDF được nối thành một dòng đúng tên miền, bỏ ngắt dòng do trình bày. Nhiệt độ thực tế thay đổi theo thời gian và không cố định ở giá trị trong ảnh mẫu.

Đáp án các chỗ trống bài 5
ÔĐáp ánÝ nghĩa
1asyncKhai báo hàm bất đồng bộ xemThoiTiet.
2tryBao phần gọi API trong khối bắt lỗi.
3awaitChờ fetch lấy tọa độ.
4awaitChờ đọc JSON từ response tọa độ.
5awaitChờ fetch lấy thời tiết theo tọa độ.
6catchXử lý lỗi mạng, dữ liệu hoặc thành phố không tìm thấy.
7errLấy thông báo từ err.message.
Chạy bài 5 →Tải mã nguồn
Xem mã hoàn chỉnh
// (1) Hàm bất đồng bộ.
async function xemThoiTiet() {
  const city = document.getElementById("city").value.trim();
  const box = document.getElementById("box");
  const button = document.getElementById("weather-button");
  if (!city) {
    alert("Vui lòng nhập tên thành phố!");
    return;
  }
  button.disabled = true;
  box.style.display = "block";
  box.className = "loading";
  box.textContent = "Đang tải dữ liệu...";
  // (2) Khối bắt lỗi cho hai yêu cầu mạng.
  try {
    const geoURL = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1&language=vi`;
    // (3) Chờ response; (4) chờ đọc JSON.
    const geoRes = await fetch(geoURL);
    if (!geoRes.ok) throw new Error(`Không tải được tọa độ (HTTP ${geoRes.status})`);
    const geoData = await geoRes.json();
    if (!geoData.results || geoData.results.length === 0) {
      throw new Error(`Không tìm thấy thành phố "${city}"`);
    }
    const { latitude, longitude, name, country } = geoData.results[0];
    const weatherURL = `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current_weather=true`;
    // (5) Chờ phản hồi thời tiết.
    const weatherRes = await fetch(weatherURL);
    if (!weatherRes.ok) throw new Error(`Không tải được thời tiết (HTTP ${weatherRes.status})`);
    const weatherData = await weatherRes.json();
    const { temperature, windspeed } = weatherData.current_weather;
    const locationName = document.createElement("strong");
    locationName.textContent = `${name}, ${country}`;
    const temp = document.createElement("span");
    temp.className = "temp";
    temp.textContent = `${temperature}°C`;
    const wind = document.createTextNode(`Gió: ${windspeed} km/h`);
    const coordinates = document.createElement("small");
    coordinates.className = "coordinates";
    coordinates.textContent = `Tọa độ: ${latitude.toFixed(2)}, ${longitude.toFixed(2)}`;
    box.className = "success";
    // Dùng textContent để tên từ API luôn là văn bản thuần.
    box.replaceChildren(locationName, document.createElement("br"), temp,
      document.createElement("br"), wind, document.createElement("br"), coordinates);
  // (6) Bắt lỗi; (7) lấy err.message.
  } catch (err) {
    box.className = "error";
    box.textContent = "Lỗi: " + err.message;
  } finally {
    button.disabled = false;
  }
}

document.getElementById("weather-form").addEventListener("submit", event => {
  event.preventDefault();
  xemThoiTiet();
});

UserCard và props

UserCard nhận name và age qua props. App truyền name="Alice", age={22}; kết quả là Tên: Alice và Tuổi: 22.

Đáp án các chỗ trống bài 6

2nameHiển thị giá trị prop name.
3ageHiển thị giá trị prop age.

import React from "react";

function UserCard({ name, age }) {
  return (
    <div>

    </div>
  );
}

function App() {
  return (
    <div>
      <UserCard name="Alice" age={22} />
    </div>
  );

Bài tập 07

Counter với useState

useState(0) tạo count và setCount. Cập nhật dạng prev => prev ± 1 sử dụng state trước đó, tránh phụ thuộc vào giá trị trong closure khi React gom nhiều cập nhật.

Đáp án các chỗ trống bài 7
ÔĐáp ánÝ nghĩa
1useStateImport Hook quản lý state.
2countGiá trị đếm hiện tại.
3setCountHàm cập nhật giá trị đếm.
4useStateKhởi tạo state bằng 0.
5prevTham số của hàm cập nhật, nhận state trước đó.
Chạy bài 7 →Tải mã nguồn
Xem mã hoàn chỉnh
import React, { useState } from "react";

function Counter() {
  const [count, setCount] = useState(0);
  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(prev => prev + 1)}>Tăng</button>{" "}
      <button onClick={() => setCount(prev => prev - 1)}>Giảm</button>
    </div>
  );
}

export { Counter };
export default Counter;

Bài tập 08

Tải người dùng bằng useEffect

useEffect với [] tải dữ liệu khi component mount, đọc res.json() rồi setUsers(data). Danh sách dùng user.id làm key. [] không có nghĩa chỉ chạy một lần trong suốt đời ứng dụng: khi component mount lại Effect chạy lại; StrictMode có chu kỳ kiểm tra bổ sung trong môi trường phát triển.

Đáp án các chỗ trống bài 8
ÔĐáp ánÝ nghĩa
1useEffectImport Hook thực thi tác vụ sau render.
2useEffectĐăng ký Effect tải danh sách người dùng.
3jsonĐọc nội dung Response thành JSON.
4setUsersCập nhật state bằng dữ liệu nhận được.
5[]Mảng phụ thuộc rỗng: Effect chạy khi component mount.
Chạy bài 8 →Tải mã nguồn
Xem mã hoàn chỉnh
import React, { useState, useEffect } from "react";

function UserList() {
  const [users, setUsers] = useState([]);
  useEffect(() => {
    fetch("https://jsonplaceholder.typicode.com/users")
      .then(res => res.json())
      .then(data => setUsers(data));
  }, []);
  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

export { UserList };
export default UserList;

Bài tập 09

Home và About với React Router

BrowserRouter cung cấp ngữ cảnh định tuyến. Link dùng to, Routes chứa Route, Route dùng path để hiển thị Home hoặc About. Bản đáp án giữ đường dẫn / và /about; demo xuất bản thêm basename để không chiếm trang chủ môn học.

Đáp án các chỗ trống bài 9
ÔĐáp ánÝ nghĩa
1BrowserRouterRouter dựa trên History API của trình duyệt.
2toThuộc tính đích điều hướng của Link.
3RoutesChứa các khai báo Route.
4pathĐường dẫn cần khớp của Route.
5AboutComponent hiển thị cho đường dẫn /about.
Chạy bài 9 →Tải mã nguồn
Xem mã hoàn chỉnh
import React from "react";
import { BrowserRouter, Routes, Route, Link } from "react-router-dom";

function Home() {
  return <h2>Home</h2>;
}

function About() {
  return <h2>About</h2>;
}

function App() {
  return (
    <BrowserRouter>
      <nav>
        <Link to="/">Home</Link>{" "}
        <Link to="/about">About</Link>
      </nav>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
      </Routes>
    </BrowserRouter>
  );
}

export default App;

Bài tập 10

Chia sẻ giao diện bằng Context

createContext("light") đặt giá trị mặc định. ThemeContext.Provider value="dark" ghi đè cho cây con; useContext(ThemeContext) trong ThemedButton nhận dark, nên nút hiện Giao diện hiện tại: dark.

Đáp án các chỗ trống bài 10
ÔĐáp ánÝ nghĩa
1createContextTạo context có giá trị mặc định light.
2ProviderCung cấp giá trị dark cho cây con.
3useContextĐọc giá trị context gần nhất.
4ThemeContextContext cần tiêu thụ.
5themeGiá trị được đọc từ Provider để hiển thị trên nút.
Chạy bài 10 →Tải mã nguồn
Xem mã hoàn chỉnh
import React, { createContext, useContext } from "react";

const ThemeContext = createContext("light");

function App() {
  return (
    <ThemeContext.Provider value="dark">
      <ThemedButton />
    </ThemeContext.Provider>
  );
}

function ThemedButton() {
  const theme = useContext(ThemeContext);
  return <button>Giao diện hiện tại: {theme}</button>;
}

export { ThemeContext, ThemedButton };
export default App;
